cd /etc/httpd && mkdir ssl ; cd ssl
openssl genrsa 2048 > private.key
openssl rsa -in private.key -pubout > public.key
openssl req -new -key private.key > request.csr
openssl req -text -in request.csr

# you send this request.csr to CA


-----CA-terminal-----

# the private key is specific to CA (but for test we create self-signed certificate in the same vm with the samep private key)
openssl x509 -in request.csr -out certificate.crt -req -signkey private.key -days 365

openssl x509 -text -in certificate.crt  # get info of the certificate
openssl x509 -pubkey -noout -in certificate.crt   # get pubkey within certificate(this is always the exact public key you created in your vm at first)

vim /etc/httpd/conf.d/ssl.conf
```
SSLCertificateFile /etc/httpd/ssl/certificate.crt
...
SSLCertificateKeyFile /etc/httpd/ssl/private.key
```
