openssl genrsa -out private.key 2048

openssl rsa -in private.key -pubout -out public.key
openssl req -new -key private.key -out request.csr
openssl req -text -in request.csr -noout -verify

# create a request.csr in one step without any prompt:
openssl req -new -key private.key -out request.csr \
-subj "/C=US/ST=Utah/L=Lehi/O=Your Company, Inc./OU=IT/CN=yourdomain.com"

# create both private.key and request.csr in one step (no reqeust.csr prompt):
openssl req -new \
-newkey rsa:2048 -nodes -keyout private.key \
-out request.csr \
-subj "/C=US/ST=Utah/L=Lehi/O=Your Company, Inc./OU=IT/CN=yourdomain.com"

openssl genrsa 2048 > private.key
openssl rsa -in private.key -pubout > public.key
openssl req -new -key private.key > request.csr
openssl req -text -in request.csr

# you send this request.csr to CA

# the private key is specific to CA (but for test we create self-signed certificate in the same vm with the same private key)
openssl x509 -in request.csr -out certificate.crt -req -signkey private.key -days 365

openssl x509 -text -in certificate.crt -noout  # get info of the certificate
openssl x509 -pubkey -in certificate.crt -noout  # get pubkey within certificate(this is always the exact public key you created in your vm at first)

# convert PEM certificate format to PKCS#12 
```
PKCS#12: 
an archival file that stores both the certificate and the private key and its 
useful when you migrate certificates and keys from one system to another as it 
contains all necessary files.
```
# you will be asked to encrypt the file with strong password.
# this is a good practice when migrate cert to another system
# it combines your private key and your certificate into a single .pfx file.
openssl pkcs12 -export -name "exampleDomain-selfsigned-(365days)" -out certificate.pfx -inkey private.key -in certificate.crt 

# PKCS#12 to PEM
openssl pkcs12 -in certificate.pfx -nocerts -out private.key -nodes # extract private key
openssl pkcs12 -in certificate.pfx -nokeys -clcerts -out certificate.crt # extract certificate.crt

# PEM to DER
openssl x509 -inform PEM -in certificate.crt -outform DER -out certificate.der  # convert certificate.crt to DER
openssl rsa -inform PEM -in private.key -outform DER -out private_key.der  # convert private.key to DER

# DER to PEM
openssl x509 -inform DER -in certificate.der -outform PEM -out certificate_PEM.crt # certificate back to PEM 
openssl rsa -inform DER -in private_key.der -outform PEM -out private_PEM.key # private key back to PEM
